import { MetadataRoute } from 'next'

export default function robots(): MetadataRoute.Robots {
    return {
        rules: [
            {
                userAgent: '*',
                allow: '/',
                disallow: [
                    '/api/',
                    '/admin/',
                    '/dashboard/',
                    '/_next/',
                    '/private/',
                    '*.json$',
                ],
            },
            {
                userAgent: 'GPTBot',
                disallow: '/',
            },
            {
                userAgent: 'Google-Extended',
                disallow: '/',
            },
            {
                userAgent: 'CCBot',
                disallow: '/',
            }
        ],
        sitemap: 'https://thecyberhub.org/sitemap.xml',
        host: 'https://thecyberhub.org',
    }
}
